#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MachineError {
    pub code: String,
    pub message: String,
    pub deleted: Option<bool>,
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MachineProblem {
    pub kind: ProblemKind,
    pub code: String,
    pub message: String,
}
pub trait MachineServer: Send + Sync + 'static {
    fn start(
        &self,
        input: Option<Value>,
        request_id: String,
    ) -> impl Future<Output = Result<ServerAnswer, TransportError>> + Send;
    fn event(
        &self,
        request: EventRequest,
    ) -> impl Future<Output = Result<ServerAnswer, TransportError>> + Send;
    fn get(
        &self,
        id: String,
    ) -> impl Future<Output = Result<ServerAnswer, TransportError>> + Send;
    fn subscribe(&self, id: String, sink: StreamSink) -> Box<dyn Subscription>;
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum MachineStatus {
    Loading,
    Ready,
    Stopped,
}
#[derive(Clone, Debug, PartialEq)]
pub struct MachineView {
    pub status: MachineStatus,
    pub id: Option<String>,
    pub snapshot: Option<Snapshot>,
    pub confirmed: Option<Snapshot>,
    pub row_id: Option<String>,
    pub revision: Option<i64>,
    pub due_at: Option<String>,
    pub pending: bool,
    pub problem: Option<MachineProblem>,
    pub error: Option<MachineError>,
    pub unsaved: bool,
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct ManifestDocument {
    pub id: String,
    pub operation: String,
    pub document: String,
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ProblemKind {
    Rejected,
    Transport,
}
pub type Provided<T> = Pin<Box<dyn Future<Output = Result<T, TransportError>> + Send>>;
pub type RequestHeaders = Arc<dyn Fn() -> Provided<HeaderMap> + Send + Sync>;
#[derive(Clone, Debug, PartialEq)]
pub enum SendOutcome {
    Confirmed(ServerSnapshot),
    Refused(MachineProblem),
    Failed(MachineProblem),
    Stopped(MachineError),
}
#[derive(Clone, Debug, PartialEq)]
pub enum ServerAnswer {
    Stored { id: Option<String>, snapshot: ServerSnapshot },
    Refused(ServerProblem),
}
pub struct ServerMachineController {}
impl ServerMachineController {
    pub fn new<S: MachineServer>(
        options: ServerMachineOptions<S>,
    ) -> Result<Self, IrError> {}
    pub fn start(&self) -> impl std::future::Future<Output = ()> + Send + 'static {}
    pub fn send(
        &self,
        trigger: impl Into<String>,
        input: Option<Value>,
    ) -> impl std::future::Future<Output = SendOutcome> + Send + 'static {}
    pub fn sync_to(
        &self,
        target: impl Into<String>,
        fields: Option<PlanEvent>,
        navigation: Vec<String>,
    ) -> impl std::future::Future<Output = SyncOutcome> + Send + 'static {}
    pub fn retry(
        &self,
    ) -> impl std::future::Future<Output = SendOutcome> + Send + 'static {}
    pub fn view(&self) -> Arc<MachineView> {}
    pub fn watch(&self) -> watch::Receiver<Arc<MachineView>> {}
    pub fn can_send(&self, trigger: &str, input: Option<&Value>) -> bool {}
    pub fn available_triggers(&self) -> Vec<String> {}
    pub fn dispose(&self) {}
}
impl Drop for ServerMachineController {}

    pub retry_delays: Option<Vec<Duration>>,
    pub runtime: Option<tokio::runtime::Handle>,
}
impl<S> ServerMachineOptions<S> {
    pub fn new(ir: Value, ir_hash: impl Into<String>, server: S) -> Self {}
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct ServerProblem {
    pub code: String,
    pub message: String,
}
#[derive(Clone, Debug, PartialEq)]
pub struct ServerSnapshot {
    pub row_id: Option<String>,
    pub revision: Option<i64>,
    pub state: String,
    pub context: Map,
    pub due_at: DueAt,
}
#[derive(Clone, Debug, PartialEq)]
pub enum StreamMessage {
    Changed(ServerSnapshot),
    Gone { not_found: bool },
    Reload,
    ClientOutdated,
    Refused(ServerProblem),
}
pub type StreamSink = std::sync::Arc<dyn Fn(StreamMessage) + Send + Sync>;
pub trait Subscription: Send {}
#[derive(Clone, Debug, PartialEq)]
pub enum SyncOutcome {
    Synced { snapshot: Snapshot, revision: Option<i64> },
    Superseded,
    Unreachable,
    Exceeded,
    Invalid(PlanInvalid),
    Refused(MachineProblem),
    Failed(MachineProblem),
    Stopped(MachineError),
}
