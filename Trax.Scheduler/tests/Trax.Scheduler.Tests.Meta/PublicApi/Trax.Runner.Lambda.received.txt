{

        protected abstract void ConfigureServices(Microsoft.Extensions.DependencyInjection.IServiceCollection services, Microsoft.Extensions.Configuration.IConfiguration configuration);
        public System.Threading.Tasks.Task<object?> FunctionHandler(Trax.Scheduler.Services.Lambda.LambdaEnvelope envelope, Amazon.Lambda.Core.ILambdaContext context) { }
        public System.Threading.Tasks.Task RunLocalAsync(string[] args) { }
        public System.Threading.Tasks.Task RunLocalAsync(string[] args, System.Threading.CancellationToken cancellationToken) { }
    }
}
