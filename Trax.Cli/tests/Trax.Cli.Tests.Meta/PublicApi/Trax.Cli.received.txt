namespace Trax.Cli.Models
{
    public class ApiEnum
    {
        public ApiEnum() { }
        public string? Description { get; init; }
        public required string Name { get; init; }
        public required System.Collections.Generic.List<string> Values { get; init; }
    }
    public class ApiField
    {
        public ApiField() { }
        public string? Description { get; init; }
        public bool IsNullable { get; init; }
        public bool IsRequired { get; init; }
        public required string Name { get; init; }
        public required string TypeName { get; set; }
    }
    public class ApiOperation
    {
        public ApiOperation() { }
        public string? Description { get; init; }
        public string? Group { get; init; }
        public string? HttpMethod { get; init; }
        public string? HttpPath { get; init; }
        public required Trax.Cli.Models.ApiType InputType { get; init; }
        public required Trax.Cli.Models.OperationKind Kind { get; init; }
        public required string Name { get; init; }
        public required Trax.Cli.Models.ApiType OutputType { get; init; }
    }
    public class ApiSchema
    {
        public ApiSchema() { }
        public System.Collections.Generic.List<Trax.Cli.Models.ApiEnum> Enums { get; init; }
        public System.Collections.Generic.List<Trax.Cli.Models.ApiOperation> Operations { get; init; }
        public required string SchemaType { get; init; }
        public required string SourceFile { get; init; }
        public System.Collections.Generic.List<Trax.Cli.Models.ApiType> Types { get; init; }
    }
    public class ApiType
    {
        public ApiType() { }
        public System.Collections.Generic.List<Trax.Cli.Models.ApiField> Fields { get; init; }
        public bool IsBuiltIn { get; init; }
        public required string Name { get; init; }
    }
    public enum OperationKind
    {
        Query = 0,
        Mutation = 1,
    }
}
